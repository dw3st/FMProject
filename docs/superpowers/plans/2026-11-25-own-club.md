# Fundar o próprio clube — plano de implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** depois de todas as conquistas, o técnico refunda um clube da última divisão de um país como clube dele, sendo técnico e presidente (report 17d9bdee).

**Architecture:** mesmo `squadId`, nova identidade (nome, sigla, cores, elenco gerado). As regras puras ficam em `src/Domain/ownClub/` (liberação, elenco, preço do ingresso, salário). A fundação é uma rota (`src/backend/ownClubRoutes.ts` + `ownClubWorld.ts`) que reaproveita `resignFromClub` e `acceptJobOffer` com uma oferta sintética. O "presidente" é uma marca no elenco (`Squad.foundedBy`), lida nos poucos pontos onde a diretoria decidia.

**Tech Stack:** Bun, TypeScript, React 19, bun:test.

**Desvios da spec:** a liberação conta todas as conquistas do catálogo de hoje (31 trilhas × 3 + 18 secretas = 111; a spec dizia 108, a contagem antiga). A spec chama o campo de `Squad.founded`, mas esse nome já existe (o ano de fundação, um número). Aqui ele se chama `Squad.foundedBy: { seatId; on }`.

```ts
import { SECRETS, TIERS, TRACKS } from "@/Domain/achievements/catalog";
import { managerWeeklyWage } from "@/Domain/managers/managerContract";
import { FOUNDED } from "@/Domain/ownClub/ownClubConfig";
import type { AchievementsState } from "@/types/achievementTypes";
import type { Squad } from "@/types/playerTypes";

/** Every key that must be unlocked: every tier of every track and every secret (111 today). */
export const OWN_CLUB_KEYS: string[] = [
  ...Object.keys(TRACKS).flatMap((t) => TIERS.map((tier) => `${t}:${tier}`)),
  ...Object.keys(SECRETS).map((s) => `secret:${s}`),
];

export function ownClubUnlock(state: AchievementsState | undefined): { unlocked: boolean; done: number; total: number } {
  const total = OWN_CLUB_KEYS.length;
  const unlocked = state?.unlocked ?? {};
  const done = OWN_CLUB_KEYS.filter((k) => (unlocked as Record<string, string | undefined>)[k]).length;
  return { unlocked: done === total, done, total };
}

export interface ClubIdentity { name: string; shortName: string; colors: [string, string] }

const HEX = /^#[0-9a-fA-F]{6}$/;
const CONTROL = /[\u0000-\u001f\u007f]/;

export function parseIdentity(body: unknown): { ok: true; identity: ClubIdentity } | { ok: false } {
  if (!body || typeof body !== "object") return { ok: false };
  const { name, shortName, colors } = body as Record<string, unknown>;
  if (typeof name !== "string" || name !== name.trim() || CONTROL.test(name)) return { ok: false };
  if (name.length < FOUNDED.NAME_MIN || name.length > FOUNDED.NAME_MAX) return { ok: false };
  if (typeof shortName !== "string" || !/^[A-Za-z]{3}$/.test(shortName)) return { ok: false };
  if (!Array.isArray(colors) || colors.length !== 2 || !colors.every((c) => typeof c === "string" && HEX.test(c))) return { ok: false };
  if (colors[0].toLowerCase() === colors[1].toLowerCase()) return { ok: false };
  return { ok: true, identity: { name, shortName: shortName.toUpperCase(), colors: [colors[0], colors[1]] } };
}

export interface FoundedSlot { line: "GK" | "DEF" | "MID" | "FWD"; level: number; age: number; i: number }

/** The 22 players of the refounded club: by line, ages spread over the range, all at the same level. */
export function foundedSquadPlan(worstLevel: number): FoundedSlot[] {
  const level = worstLevel - FOUNDED.LEVEL_GAP;
  const out: FoundedSlot[] = [];
  let i = 0;
  const span = FOUNDED.AGE_MAX - FOUNDED.AGE_MIN;
  for (const [line, n] of Object.entries(FOUNDED.LINES) as ["GK" | "DEF" | "MID" | "FWD", number][]) {
    for (let k = 0; k < n; k++) {
      out.push({ line, level, age: FOUNDED.AGE_MIN + Math.round((span * k) / Math.max(1, n - 1)), i });
      i++;
    }
  }
  return out;
}

export function ticketDemandMult(priceMult: number): number {
  return priceMult === 1 ? 1 : priceMult ** -FOUNDED.PRICE_ELASTICITY;
}

export function isPresident(squad: Pick<Squad, "foundedBy"> | null | undefined, seatId: string): boolean {
  return !!squad?.foundedBy && squad.foundedBy.seatId === seatId;
}

/** The president's wage ceiling: what the contract rule would pay him. */
export function managerWageCap(revenueBasis: number, reputation: number): number {
  return managerWeeklyWage(revenueBasis, reputation);
}

/** A valid ticket multiplier (0,5–2,0 in steps of 0,1). */
export function parseTicketPrice(v: unknown): number | null {
  if (typeof v !== "number" || !Number.isFinite(v)) return null;
  const r = Math.round(v * 10) / 10;
  if (Math.abs(r - v) > 1e-9 || r < FOUNDED.PRICE_MIN || r > FOUNDED.PRICE_MAX) return null;
  return r;
}
```
